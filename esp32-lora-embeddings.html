// ESP32-S3 Sender Gateway Firmware
#include <Arduino.h>
#include <SPI.h>
#include <LoRA.h>

#define SCK     12
#define MISO    13
#define MOSI    11
#define SS      10
#define RST     14
#define DI0     21
#define BAND    915E6

struct __attribute__((__packed__)) LoraPacket {
    uint8_t  msgType;
    uint16_t nodeId;
    int16_t  quantX;
    int16_t  quantY;
    int16_t  quantZ;
    uint8_t  vectorSum;
    uint8_t  padding[6];
    uint16_t crc16;
};

void setup() {
    Serial.begin(115200);
    SPI.begin(SCK, MISO, MOSI, SS);
    LoRA.setPins(SS, RST, DI0);
    if (!LoRA.begin(BAND)) {
        Serial.println("[ERROR] ESP32-S3 LoRA Init Failed!");
        while (1);
    }
    LoRA.setTxPower(20);
    Serial.println("[SYSTEM] ESP32-S3 Gateway Origin Ready.");
}

void loop() {
    if (Serial.available() >= sizeof(LoraPacket)) {
        LoraPacket packet;
        Serial.readBytes((char*)&packet, sizeof(LoraPacket));
        
        LoRA.beginPacket();
        LoRA.write((uint8_t*)&packet, sizeof(LoraPacket));
        LoRA.endPacket();
        
        Serial.print("[TX SUCCESS] ESP32-S3 Transmitted Packet ID: ");
        Serial.println(packet.nodeId);
    }
}

// ESP32-S3 Receiver Remote Node Firmware
#include <Arduino.h>
#include <SPI.h>
#include <LoRA.h>

#define SCK     12
#define MISO    13
#define MOSI    11
#define SS      10
#define RST     14
#define DI0     21
#define BAND    915E6

struct __attribute__((__packed__)) LoraPacket {
    uint8_t  msgType;
    uint16_t nodeId;
    int16_t  quantX;
    int16_t  quantY;
    int16_t  quantZ;
    uint8_t  vectorSum;
    uint8_t  padding[6];
    uint16_t crc16;
};

void setup() {
    Serial.begin(115200);
    SPI.begin(SCK, MISO, MOSI, SS);
    LoRA.setPins(SS, RST, DI0);
    if (!LoRA.begin(BAND)) {
        Serial.println("[ERROR] ESP32-S3 LoRA Init Failed!");
        while (1);
    }
    Serial.println("[SYSTEM] ESP32-S3 Receiver Active. Listening...");
}

void loop() {
    int packetSize = LoRA.parsePacket();
    if (packetSize == sizeof(LoraPacket)) {
        LoraPacket packet;
        LoRA.readBytes((uint8_t*)&packet, sizeof(LoraPacket));
        
        float x = (packet.quantX / 32767.0f) * 100.0f;
        float y = (packet.quantY / 32767.0f) * 100.0f;
        float z = (packet.quantZ / 32767.0f) * 100.0f;
        
        Serial.print("[RX ESP32-S3] Node:"); Serial.print(packet.nodeId);
        Serial.print(" | X:"); Serial.print(x);
        Serial.print(" Y:"); Serial.print(y);
        Serial.print(" Z:"); Serial.print(z);
        Serial.print(" | RSSI:"); Serial.print(LoRA.packetRssi());
        Serial.print(" SNR:"); Serial.println(LoRA.packetSnr());
    }
}
